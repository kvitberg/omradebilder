name: Deploy til GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

# Pages-deploy trenger disse rettighetene på GITHUB_TOKEN.
permissions:
  contents: read
  pages: write
  id-token: write

# Én deploy om gangen; nye pushes venter i stedet for å avbryte.
concurrency:
  group: pages
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm

      - name: Installer avhengigheter
        run: npm ci

      - name: Bygg statisk side
        # Prosjektsider serveres fra /<repo>/, så alle stier må ha det foran.
        env:
          NEXT_PUBLIC_BASE_PATH: /${{ github.event.repository.name }}
        run: npm run build

      - uses: actions/configure-pages@v5

      - uses: actions/upload-pages-artifact@v3
        with:
          path: ./out

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
